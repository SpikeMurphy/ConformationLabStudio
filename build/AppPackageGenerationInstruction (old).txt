# install colabfold in python environment (e.g. colabfold)
# use shell skript installer

# activate python enwironment
conda activate shipfold

# install pywebview for mol*viewer
pip install pywebview

# install psutil for system stats (CPU/RAM)
pip install psutil

# install tkinterdnd2 for drag and drop (not done for M1)
pip install tkinterdnd2

# install packing package
conda install -c conda-forge conda-pack

# go to where the environment pack should be saved
# e.g. cd '/Users/spikemurphymuller/ShipFoldGeneration'
cd '<path>'

# pack environment into tar.gz
conda pack -n shipfold -o shipfold_env.tar.gz

# copy GUI and depending data in non iColud folder 
# e.g. '/Users/spikemurphymuller/ShipFoldGeneration'
# (icon-file, python-skript, run_molstar.py, shipfold_env.tar.gz, spect-file, logo-file)

# install pyinstaller
conda install -c conda-forge pyinstaller

# go to where the python GUI is saved
# e.g. cd '/Users/spikemurphymuller/ShipFoldGeneration'
cd '<path>'

#===== first application  generation =====
# first application generation only
pyinstaller \
--windowed \
--onefile \
--name "ShipFoldStudio" \
--icon "ShipFoldIcon.icns" \
--add-data "shipfold_env.tar.gz:." \
--add-data "ShipFoldBackground.png:." \
--add-data "ShipFoldLogo.png:." \
ShipFoldStudio_v1.6.py
#============================================

# spect datei anpassen mit versionsdaten
# spect datei vor ort belassen

#===== further application  generation =====
# activate python enwironment
conda activate shipfold

# go to where the environment pack should be saved
# e.g. cd '/Users/spikemurphymuller/ShipFoldGeneration'
cd '<path>'

# reset
rm -rf build dist

# generate application
pyinstaller ShipFoldStudio.spec

#============================================
